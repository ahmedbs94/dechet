# Lancement du worker IA (depuis backend/) :
#   python -m ai_worker.worker
#
# Ou via Docker (voir ai_worker/Dockerfile) :
#   docker build -f ai_worker/Dockerfile -t ecorewind-ai-worker .
#   docker run --env-file .env ecorewind-ai-worker
